for (i in 1:n) {
  # Training indices (all except i) and testing index (just i)
  train_idx <- -i
  test_idx <- i
  
  # 1. Random Forest
  # randomForest expects a standard matrix/data.frame; drop=FALSE keeps it 2D
  rf_model <- randomForest(
    x = as.matrix(twdfmk[train_idx, ]), 
    y = y[train_idx], 
    ntree = 1000
  )
  pred_rf[i] <- as.character(predict(rf_model, newdata = as.matrix(twdfmk[test_idx, , drop = FALSE])))
  
  # 2. Naive Bayes (quanteda.textmodels)
  nb_model <- textmodel_nb(twdfmk[train_idx, ], y[train_idx])
  pred_nb[i] <- as.character(predict(nb_model, newdata = twdfmk[test_idx, ], type = "class"))
  
  # 3. Linear SVM (quanteda.textmodels)
  svm_model <- textmodel_svmlin(twdfmk[train_idx, ], y[train_idx])
  pred_svm[i] <- as.character(predict(svm_model, newdata = twdfmk[test_idx, ]))
}

cat("Running LOOCV loop across", n, "documents...\n")
for (i in 1:n) {
  # Training indices (all except i) and testing index (just i)
  train_idx <- -i
  test_idx <- i
  
  # 1. Random Forest
  # randomForest expects a standard matrix/data.frame; drop=FALSE keeps it 2D
  rf_model <- randomForest(
    x = as.matrix(twdfmk[train_idx, ]), 
    y = y[train_idx], 
    ntree = 1000
  )
  pred_rf[i] <- as.character(predict(rf_model, newdata = as.matrix(twdfmk[test_idx, , drop = FALSE])))
  
  # 2. Naive Bayes (quanteda.textmodels)
  nb_model <- textmodel_nb(twdfmk[train_idx, ], y[train_idx])

# LLM generated LOOCV code # 
# Total number of documents
n <- nrow(twdfmk)

# Ensure the target variable is a factor
y <- factor(western)

# Vectors to store the predicted classes for each model
pred_rf  <- character(n)
pred_nb  <- character(n)
pred_svm <- character(n)

# Start LOOCV loop
cat("Running LOOCV loop across", n, "documents...\n")
for (i in 1:n) {
  # Training indices (all except i) and testing index (just i)
  train_idx <- -i
  test_idx <- i
  
  # 1. Random Forest
  # randomForest expects a standard matrix/data.frame; drop=FALSE keeps it 2D
  rf_model <- randomForest(
    x = as.matrix(twdfmk[train_idx, ]), 
    y = y[train_idx], 
    ntree = 1000
  )
  pred_rf[i] <- as.character(predict(rf_model, newdata = as.matrix(twdfmk[test_idx, , drop = FALSE])))
  
  # 2. Naive Bayes (quanteda.textmodels)
  nb_model <- textmodel_nb(twdfmk[train_idx, ], y[train_idx])
  pred_nb[i] <- as.character(predict(nb_model, newdata = twdfmk[test_idx, ], type = "class"))
  
  # 3. Linear SVM (quanteda.textmodels)
  svm_model <- textmodel_svmlin(twdfmk[train_idx, ], y[train_idx])
  pred_svm[i] <- as.character(predict(svm_model, newdata = twdfmk[test_idx, ]))
}

# --- Evaluation (Accuracy) ---
actual <- as.character(y)
acc_rf  <- mean(pred_rf == actual)
acc_nb  <- mean(pred_nb == actual)
acc_svm <- mean(pred_svm == actual)

cat("\n--- LOOCV Accuracies ---\n")
cat(sprintf("Random Forest: %.2f%%\n", acc_rf * 100))
cat(sprintf("Naive Bayes:   %.2f%%\n", acc_nb * 100))
cat(sprintf("Linear SVM:    %.2f%%\n", acc_svm * 100))
