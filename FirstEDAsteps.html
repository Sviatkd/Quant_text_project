data = bind_rows(tibble(document = "CIS2011", text = text),
tibble(document ="Kyrgyz", text = textk),
tibble(document = "FTA1994", text = textFTA1994),
tibble(document = "Georgia", text = textGeorgia),
tibble(document = "Armenia", text = textArmenia))
data_combined <- data %>%
  group_by(document) %>%
  summarise(text = paste(text, collapse = "\n"))
Agreement_corpus1 = corpus(data_combined,text_field = "text", docid_field = "document")
